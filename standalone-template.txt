async function shrink(file){
  const url=URL.createObjectURL(file);
  try{
    const img=await new Promise((res,rej)=>{const i=new Image();i.onload=()=>res(i);i.onerror=rej;i.src=url});
    const side=Math.min(img.width,img.height),out=Math.min(512,side);
    const c=document.createElement("canvas");c.width=c.height=out;
    c.getContext("2d").drawImage(img,(img.width-side)/2,(img.height-side)/2,side,side,0,0,out,out);
    const b=await new Promise(r=>c.toBlob(r,"image/webp",.92));
    return b&&b.type==="image/webp"?b:file;
  }finally{URL.revokeObjectURL(url)}
}

async function addPngText(blob,key,text){
  const buf=new Uint8Array(await blob.arrayBuffer());
  const enc=s=>Uint8Array.from(s,c=>c.charCodeAt(0)&255);
  const body=new Uint8Array([...enc(key),0,...enc(text)]);
  const type=enc("tEXt"),len=body.length;
  const chunk=new Uint8Array(12+len),dv=new DataView(chunk.buffer);
  dv.setUint32(0,len);chunk.set(type,4);chunk.set(body,8);
  const crcIn=new Uint8Array(4+len);crcIn.set(type,0);crcIn.set(body,4);dv.setUint32(8+len,CRC(crcIn));
  const iend=buf.length-12;const out=new Uint8Array(buf.length+chunk.length);
  out.set(buf.subarray(0,iend),0);out.set(chunk,iend);out.set(buf.subarray(iend),iend+chunk.length);
  return new Blob([out],{type:"image/png"});
}

/* 用 GMT+8（UTC+8）計週期，不管部機設了什麼時區 */
function rKey(p,now){
  if(p==="o")return "o";
  const t=new Date((now??Date.now())+(8-rHour())*36e5);const y=t.getUTCFullYear(),mo=t.getUTCMonth(),d=t.getUTCDate();
  if(p==="d")return `d${y}-${mo+1}-${d}`;
  if(p==="w"){const m=new Date(Date.UTC(y,mo,d-(t.getUTCDay()+6)%7));return `w${m.getUTCFullYear()}-${m.getUTCMonth()+1}-${m.getUTCDate()}`}
  if(p==="m")return `m${y}-${mo+1}`;
  if(p==="h")return `h${y}-${mo+1}-${d<=15?1:2}`;
  if(p==="b")return `b${y}-${Math.floor(mo/2)}`;
  return "o";
}
/* 自訂重置時間（GMT+8，例如 05:59、17:59）：以最近已過的重置點為週期 */
function tKey(ts,now){
  const n=now??Date.now(),HK=8*36e5,DAY=864e5,dd=Math.floor((n+HK)/DAY);let best=null;
  for(const di of[dd-1,dd])for(const z of ts){const[h,m]=z.split(":").map(Number);const b=di*DAY+h*36e5+m*6e4-HK;if(b<=n&&(best===null||b>best))best=b}
  return best===null?"t0":"t"+best;
}

function renderWeek(){
  if(!$("calWeek"))return;
  const start=CAL.wk,days=[0,1,2,3,4,5,6].map(i=>addDays(start,i)),end=days[6],today=hkToday();
  $("wkTitle").textContent=`${fmtMD(start)} – ${fmtMD(end)}`;

  const segs=[];
  for(const e of visEv()){
    const en=endOf(e);
    if(!(e.date<=end&&en>=start))continue;

    if(c0<0||c1<0)continue;
    segs.push({e,c0,c1,cl:e.date<start,cr:en>end});
  }
  segs.sort((a,b)=>a.c0-b.c0||b.c1-a.c1);
  const lanes=[];
  for(const s of segs){let L=0;while(lanes[L]!==undefined&&lanes[L]>=s.c0)L++;lanes[L]=s.c1;s.lane=L}
  const H=Math.max(1,lanes.length)*32+12;
  let h=`<div class="wkcols">`+days.map(d=>`<div class="wkcol${d===today?" tday":""}"></div>`).join("")+`</div>`;
  h+=`<div class="wkstack" style="height:${H}px">`;
  for(const s of segs){
    const e=s.e,span=s.c1-s.c0+1,k=kindOf(e),t=esc(e.title);
    h+=`<button type="button" class="wkbar k-${k}${span===1?" one":""}${s.cl?" cl":""}${s.cr?" cr":""}" data-d="${esc(e.date)}" style="--kc:${evColor(e)};top:${6+s.lane*32}px;left:calc(${s.c0} * 100% / 7 + 3px);width:calc(${span} * 100% / 7 - 6px)" title="${t}"><i class="wkico"></i><span>${t}</span></button>`;

var TW={items:[]};
function twParse(raw){
  const today=hkToday(),Y0=+today.slice(0,4),M0=+today.slice(5,7);
  const yOf=m=>(m<M0-6?Y0+1:(m>M0+6?Y0-1:Y0));
  const pd=n=>String(n).padStart(2,"0"),ymd=(m,d)=>`${yOf(m)}-${pd(m)}-${pd(d)}`,hm=x=>{const[a,b]=x.split(":");return pd(a)+":"+b};
  let t=String(raw||"").replace(/[０-９]/g,c=>String.fromCharCode(c.charCodeAt(0)-65248)).replace(/[～〜~]/g,"-").replace(/：/g,":").replace(/\r/g,"");
  const out=[];let mt=null;
  const mm=t.match(/(\d{1,2})月(\d{1,2})日\s*[（(]?[週周星期]?.?[)）]?\s*(\d{1,2}:\d{2})\s*-+\s*(\d{1,2}:\d{2})\s*進行(?:停服)?維護/);
  if(mm){mt=hm(mm[3]);out.push({date:ymd(+mm[1],+mm[2]),title:"停服維護",time:hm(mm[3]),endTime:hm(mm[4]),on:true,src:"維護時間"})}
  const nm=t.match(/下次維護時間為\s*(\d{1,2})月(\d{1,2})日/);
  if(nm)out.push({date:ymd(+nm[1],+nm[2]),title:"停服維護",time:mt||"",note:"時間以官方公告為準",on:true,src:"下次維護"});
  let body=t.split(/=+\s*改版更新內容\s*=+/)[1];if(body===undefined)body=t;body=body.split(/\n=+[^=\n]+=+/)[0];
  const secs=[];let cur=null;
  for(const L of body.split("\n")){const h=L.match(/^\s*(\d{1,2})\s*[.、．]\s*(.+)$/)||L.match(/^\s*(\d{1,2})\s+(\D.+)$/);
    if(h&&!/^\s*\d{1,2}\s*月/.test(L)){cur={title:h[2].trim(),lines:[]};secs.push(cur)}else if(cur)cur.lines.push(L)}
  const D="(\\d{1,2})月(\\d{1,2})日\\s*(維護後|維護前|\\d{1,2}:\\d{2})?\\s*後?";
  const RNG=new RegExp(D+"\\s*-+\\s*"+D),ONE=new RegExp(D);
  const mk=(title,x,extra)=>{const r=x.match(RNG);
    if(r){const e={date:ymd(+r[1],+r[2]),end:ymd(+r[4],+r[5]),title,on:true};
      if(r[3])e.time=r[3]==="維護後"?"維護後":hm(r[3]);
      if(r[6]==="維護前"){e.endTime=mt&&out[0]&&out[0].date===e.end?mt:"07:30";e.note="維護前結束";}else if(r[6]&&r[6]!=="維護後")e.endTime=hm(r[6]);else e.endTime="23:59";
      return Object.assign(e,extra)}
    const o=x.match(ONE);if(o)return Object.assign({date:ymd(+o[1],+o[2]),title,time:o[3]==="維護後"?"維護後":(o[3]&&o[3]!=="維護前"?hm(o[3]):""),perm:true,on:true,note:""},extra);return null};
  for(const sc of secs){const txt=sc.lines.join("\n");const title=sc.title.replace(/^「([^」]+)」/,"$1").replace(/^【([^】]+)】/,"$1").trim();
    if(/禮包/.test(title)&&/限購時間/.test(txt)){
      for(const blk of txt.split(/\n(?=\s*【)/)){const n=blk.match(/【(.+?)】/),p=blk.match(/限購時間\s*:\s*(.+)/),pr=blk.match(/售價\s*:\s*(.+)/);
        if(n&&p){const e=mk(n[1].trim(),p[1],{});if(e){e.perm=false;e.note=pr?"售價 "+pr[1].trim():"";e.src=title;out.push(e)}}}
      continue}
    const per=txt.match(/(?:活動|召喚|開放|販售|上架)時間\s*:\s*(.+)/);
    if(per){const e=mk(title,per[1],{});if(e){
      if(/首次登入/.test(txt)){const k=txt.match(/(?:總共持續|持續|第)\s*(\d+)\s*[日天]/g);const n=k?Math.max(...k.map(v=>+v.match(/\d+/)[0])):7;e.note=`首次登入起計 ${n} 日`}
      if(e.perm)e.note="只寫了開始日，當作永久上線";
      e.src="活動時間";out.push(e);continue}}
    const any=txt.match(ONE)||sc.title.match(ONE);
    if(any){const e=mk(title,any[0],{});if(e){e.on=false;e.note=e.perm?"內文提到的日期，當作永久上線":"";e.src="內文日期";out.push(e);continue}}
    out.push({title,on:false,src:"找不到日期"});
  }
  for(const e of out){if(!e.date)continue;const dup=CAL.ev.find(x=>x.date===e.date&&(x.title===e.title||(kindOf(e)==="m"&&kindOf(x)==="m")));if(dup){e.on=false;e.dup=true}}
