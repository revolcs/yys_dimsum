  const r=await fetch("https://api.github.com"+path,{method,cache:"no-store",headers:{"Authorization":"Bearer "+GH.tok,"Accept":"application/vnd.github+json","X-GitHub-Api-Version":"2022-11-28",...(body?{"Content-Type":"application/json"}:{})},body:body?JSON.stringify(body):undefined});
  if(r.status===404&&allow404)return null;
  if(!r.ok){const e=new Error("GitHub "+r.status);e.code="github_"+r.status;throw e}
  return r.status===204?{}:r.json();
}
const ghPath=(repo,p)=>`/repos/${encodeURIComponent(GH_CFG.owner)}/${encodeURIComponent(repo)}/contents/${p.split("/").map(encodeURIComponent).join("/")}`;
async function ghGet(repo,p){const j=await ghApi("GET",ghPath(repo,p)+"?ref="+encodeURIComponent(GH_CFG.branch),null,true);const key=repo+"/"+p;if(!j){delete GH.sha[key];return null}GH.sha[key]=j.sha;return j}
async function ghPut(repo,p,b64,msg){
  const key=repo+"/"+p;
  const put=()=>ghApi("PUT",ghPath(repo,p),{message:msg,content:b64,branch:GH_CFG.branch,...(GH.sha[key]?{sha:GH.sha[key]}:{})});
  let j;try{j=await put()}catch(e){if(e.code==="github_409"||e.code==="github_422"){await ghGet(repo,p);j=await put()}else throw e}
  if(j&&j.content&&j.content.sha)GH.sha[key]=j.content.sha;return j;
}
async function ghDel(repo,p,msg){const key=repo+"/"+p;if(!GH.sha[key]&&!(await ghGet(repo,p)))return;await ghApi("DELETE",ghPath(repo,p),{message:msg,sha:GH.sha[key],branch:GH_CFG.branch},true);delete GH.sha[key]}
/* 跟 claude 儲存同一套用法的小倉庫：doc().update/set/delete、collection().add/onSnapshot */
function ghStore(){
  const M=new Map(),subs=[];
  const emit=()=>{const docs=[...M.values()].map(o=>({id:o.id,data:()=>{const c={...o};delete c.id;return c}}));subs.forEach(f=>f({docs}))};
  const doc=p=>{const[c,id]=p.split("/");return{
    set:async v=>{if(c==="meta"&&id==="accounts"&&v&&Array.isArray(v.list)){S.accts=v.list;ghQueue()}},
    update:async v=>{if(c!=="shiki")return;const o=M.get(id);if(!o){const e=new Error("not found");e.code="not_found";throw e}Object.assign(o,v);ghQueue();emit()},
    delete:async()=>{if(c==="shiki"&&M.has(id)){M.delete(id);ghQueue();emit()}},
    onSnapshot:()=>()=>{}}};
  const collection=c=>({
    add:async v=>{const id="gh"+Date.now().toString(36)+Math.random().toString(36).slice(2,6);M.set(id,{id,...v});ghQueue();emit();return{id}},
    onSnapshot:f=>{if(c==="shiki"){subs.push(f);emit()}return()=>{}}});
  return{M,doc,collection};
}

  const d={};for(const s of list){const e={};for(const a of A){const o=(a.k==="test"&&s.ownedT===undefined)?!!s.test:!!s[OKEY(a.k)];const v=s[a.k]||"";if(o||v)e[a.k]=[o?1:0,v]}if(Object.keys(e).length)d[s.id]=[s.name,e]}
  const pubStr=pub.map(o=>JSON.stringify(o)).join(",\n"),dStr=Object.entries(d).map(([k,v])=>JSON.stringify(k)+":"+JSON.stringify(v)).join(",\n");

function ghQueue(){if(!S.isGH||!GH.store)return;clearTimeout(GH.timer);ghStatus("未同步…");GH.timer=setTimeout(()=>{GH.timer=null;GH.q=GH.q.then(ghFlush)},1200)}
async function ghFlush(){
  const pk=ghPack();GH.busy++;ghStatus("同步中…");
  try{
    if(pk.pubStr!==GH.lastPub){await ghPut(GH_CFG.pub,"data/shiki.json",b64enc(pk.pubFile()),"更新式神名單");GH.lastPub=pk.pubStr}
    if(pk.privStr!==GH.lastPriv){await ghPut(GH_CFG.priv,"records.json",b64enc(pk.privFile()),"更新紀錄");GH.lastPriv=pk.privStr}
    GH.busy--;ghStatus(GH.timer?"未同步…":"已同步 ✓");
  }catch(e){GH.busy--;ghStatus("同步失敗，稍後自動重試");toast("同步失敗："+(e.code==="github_401"?"通行證過期或錯誤":e.code==="github_403"||e.code==="github_404"?"通行證沒有寫入權限":(e.code||e.message||"網絡問題")));setTimeout(ghQueue,15000)}
}

  if(GH.tok){
    try{

      const pj=await ghGet(GH_CFG.pub,"data/shiki.json");

        const e=rec&&rec.d&&rec.d[s.id];if(e)for(const[k,v]of Object.entries(e[1]||{})){const[ow,val]=Array.isArray(v)?v:[0,""];o[OKEY(k)]=!!ow;o[k]=val||null}
        st.M.set(s.id,o)}

      S.db=st;S.assets=ghAssets;S.owner=true;S.isStatic=false;

}
